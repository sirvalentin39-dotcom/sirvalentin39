const SECRET_TOKEN = "CUBE77X9MOD2026B2";
const FOLDER_NAME = "КубКрафт_Скріншоти";
const HEADERS = ["userId", "telegramNick", "status", "timestamp", "screen1", "screen2", "screen3"];

function doGet(e)  { return handleRequest(e); }
function doPost(e) { return handleRequest(e); }

function handleRequest(e) {
  try {
    const params = parseParams(e);
    const action = params.action;
    if (!action) return ok({ success: true, status: "ok" });

    const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    ensureHeaders(sheet);

    if (action === "getAll" || action === "updateStatus" || action === "manageUser") {
      if (!safeEqual(String(params.token || ""), SECRET_TOKEN)) {
        return ok({ success: false, error: "Unauthorized" });
      }
    }

    switch (action) {
      case "create":       return handleCreate(sheet, params);
      case "check":        return handleCheck(sheet, params);
      case "getAll":       return handleGetAll(sheet);
      case "updateStatus": return handleUpdateStatus(sheet, params);
      case "manageUser":   return handleManageUser(sheet, params);
      default:             return ok({ success: false, error: "Unknown action: " + action });
    }
  } catch (error) {
    return ok({ success: false, error: String(error) });
  }
}

// ---- Дії ----

function handleCreate(sheet, params) {
  const userId = String(params.userId || "");
  const telegramNick = String(params.telegramNick || "");

  if (!userId) return ok({ success: false, error: "Missing userId" });

  // Збираємо скріншоти з обох форматів: окремі поля screenshot1..3 (FormData)
  // АБО старий масив screenshots (JSON-body)
  let rawShots = [];
  if (params.screenshot1 || params.screenshot2 || params.screenshot3) {
    rawShots = [params.screenshot1 || "", params.screenshot2 || "", params.screenshot3 || ""];
  } else if (Array.isArray(params.screenshots)) {
    rawShots = params.screenshots;
  }

  const data = sheet.getDataRange().getValues();

  // Бан блокує створення заявки
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === userId && String(data[i][2]) === "banned") {
      return ok({ success: false, error: "banned" });
    }
  }

  const screenshotUrls = saveScreenshots(userId, rawShots);

  // Оновлюємо існуючу pending-заявку (інакше створюємо нову)
  let rowIndex = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === userId && String(data[i][2]) === "pending") {
      rowIndex = i + 1;
      break;
    }
  }

  // GET-fallback приходить без скріншотів: не затираємо вже збережені чеки
  const hasNewShots = screenshotUrls.some(function (u) { return u; });

  if (rowIndex !== -1) {
    if (telegramNick) sheet.getRange(rowIndex, 2).setValue(telegramNick);
    sheet.getRange(rowIndex, 3).setValue("pending");
    sheet.getRange(rowIndex, 4).setValue(new Date());
    if (hasNewShots) {
      sheet.getRange(rowIndex, 5).setValue(screenshotUrls[0]);
      sheet.getRange(rowIndex, 6).setValue(screenshotUrls[1]);
      sheet.getRange(rowIndex, 7).setValue(screenshotUrls[2]);
    }
  } else {
    sheet.appendRow([userId, telegramNick, "pending", new Date(),
      screenshotUrls[0], screenshotUrls[1], screenshotUrls[2]]);
  }

  return ok({ success: true, message: "Заявку надіслано!" });
}

function handleCheck(sheet, params) {
  const userId = String(params.userId || "");
  if (!userId) return ok({ success: false, error: "Missing userId" });

  const data = sheet.getDataRange().getValues();
  let status = "not_found";
  for (let i = data.length - 1; i >= 1; i--) {
    if (String(data[i][0]) === userId) {
      status = String(data[i][2] || "active");
      break;
    }
  }
  return ok({ success: true, status: status });
}

function handleGetAll(sheet) {
  const data = sheet.getDataRange().getValues();
  const users = [];
  for (let i = 1; i < data.length; i++) {
    if (!data[i][0]) continue;
    users.push({
      userId: String(data[i][0] || ""),
      telegramNick: String(data[i][1] || ""),
      status: String(data[i][2] || ""),
      timestamp: data[i][3] ? String(data[i][3]) : "",
      screen1: String(data[i][4] || ""),
      screen2: String(data[i][5] || ""),
      screen3: String(data[i][6] || "")
    });
  }
  return ok({ success: true, users: users });
}

function handleUpdateStatus(sheet, params) {
  const userId = String(params.userId || "");
  const newStatus = String(params.newStatus || "");
  if (!userId || !newStatus) return ok({ success: false, error: "Missing data" });

  const data = sheet.getDataRange().getValues();
  for (let i = data.length - 1; i >= 1; i--) {
    if (String(data[i][0]) === userId) {
      sheet.getRange(i + 1, 3).setValue(newStatus);
      return ok({ success: true, updated: true });
    }
  }
  return ok({ success: true, updated: false });
}

function handleManageUser(sheet, params) {
  const userId = String(params.userId || "");
  const subAction = String(params.subAction || "");
  if (!userId || !subAction) return ok({ success: false, error: "Missing parameters" });

  const data = sheet.getDataRange().getValues();
  let rowIndex = -1;
  for (let i = data.length - 1; i >= 1; i--) {
    if (String(data[i][0]) === userId) { rowIndex = i + 1; break; }
  }
  if (rowIndex === -1) {
    sheet.appendRow([userId, "@unknown", "active", new Date(), "", "", ""]);
    rowIndex = sheet.getLastRow();
  }

  if (subAction === "ban")   { sheet.getRange(rowIndex, 3).setValue("banned"); return ok({ success: true, message: "Забанено" }); }
  if (subAction === "unban") { sheet.getRange(rowIndex, 3).setValue("active"); return ok({ success: true, message: "Розбанено" }); }
  if (subAction === "custom_change") {
    const amount = parseInt(String(params.amount || "0"), 10);
    if (isNaN(amount)) return ok({ success: false, error: "Invalid amount" });
    const cmd = amount >= 0 ? "give_" + amount : "take_" + Math.abs(amount);
    sheet.getRange(rowIndex, 3).setValue(cmd);
    return ok({ success: true, message: "Команду встановлено: " + cmd });
  }
  return ok({ success: false, error: "Unknown subAction: " + subAction });
}

// ---- Допоміжні ----

function parseParams(e) {
  if (e && e.postData && e.postData.contents) {
    try {
      const body = JSON.parse(e.postData.contents);
      return Object.assign({}, (e && e.parameter) || {}, body);
    } catch (err) { /* не JSON (FormData) — беремо e.parameter */ }
  }
  return (e && e.parameter) || {};
}

function ensureHeaders(sheet) {
  if (sheet.getLastRow() === 0) sheet.appendRow(HEADERS);
}

// Повертає посилання у форматі thumbnail (drive.google.com/thumbnail?id=...&sz=...),
// бо старий формат uc?export=view&id=... Google більше НЕ віддає у тег <img>.
// Саме цей формат коректно вбудовується як прев'ю і відкривається на весь екран.
function saveScreenshots(userId, screenshots) {
  const urls = [];
  let folder;
  try {
    const folders = DriveApp.getFoldersByName(FOLDER_NAME);
    folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(FOLDER_NAME);
  } catch (err) {
    return ["", "", ""];
  }
  for (let s = 0; s < Math.min(screenshots.length, 3); s++) {
    try {
      const raw = String(screenshots[s] || "");
      if (!raw) { urls.push(""); continue; }
      const base64Data = raw.indexOf(",") !== -1 ? raw.split(",")[1] : raw;
      const blob = Utilities.newBlob(
        Utilities.base64Decode(base64Data),
        "image/png",
        "screen_" + userId + "_" + (s + 1) + ".png"
      );
      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      urls.push("https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w1600");
    } catch (err) {
      urls.push("");
    }
  }
  while (urls.length < 3) urls.push("");
  return urls;
}

function safeEqual(a, b) {
  if (a.length !== b.length) return false;
  let diff = 0;
  for (let i = 0; i < a.length; i++) diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return diff === 0;
}

function ok(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
